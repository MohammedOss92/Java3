# ==========================================
# 1. الإعدادات والبيانات الشخصية (Configuration)
# ==========================================
git config --global user.name "Your Name"       # ضبط اسم المستخدم لجميع المشاريع على الجهاز
git config --global user.email "email@example.com" # ضبط البريد الإلكتروني لجميع المشاريع على الجهاز
git config user.name "Your Name"                # ضبط اسم المستخدم للمشروع الحالي فقط
git config user.email "email@example.com"       # ضبط البريد الإلكتروني للمشروع الحالي فقط
git config --list                              # عرض جميع إعدادات Git المطبقة حالياً

# ==========================================
# 2. إنشاء واستنساخ المشاريع (Setup & Init)
# ==========================================
git init                                       # تهيئة مجلد جديد وتحويله لمستودع Git محلي
git clone <repository_url>                     # تنزيل/استنساخ مشروع موجود من GitHub إلى جهازك

# ==========================================
# 3. تتبع وحفظ التغييرات (Basic Workflow)
# ==========================================
git status                                     # عرض حالة الملفات (المعدلة، المضافة، والتجهيز للـ Commit)
git add <file_name>                            # إضافة ملف محدد لمرحلة الانتظار (Staging Area)
git add .                                      # إضافة جميع الملفات التائهة والمعدلة لمرحلة الانتظار
git commit -m "Your commit message"            # حفظ التغييرات الحالية مع كتابة وصف للتعديل
git commit -am "Your commit message"           # إضافة التعديلات وحفظها بمرحلة واحدة (للملفات المتبعة سابقاً)

# ==========================================
# 4. المزامنة والربط مع GitHub (Remote)
# ==========================================
git remote add origin <repository_url>         # ربط المستودع المحلي بمستودع بعيد على GitHub
git remote -v                                  # عرض روابط المستودعات البعيدة المرتبطة بالمشروع
git push origin <branch_name>                  # رفع الكود والتغييرات من الفرع المحلي إلى GitHub
git pull                                       # جلب التحديثات الجديدة من GitHub ودمجها تلقائياً مع كودك المحلي
git fetch                                      # جلب البيانات والتغييرات من GitHub دون دمجها مع كودك المحلي

# ==========================================
# 5. إدارة الفروع (Branches)
# ==========================================
git branch                                     # عرض جميع الفروع المحلية وتحديد الفرع الحالي
git branch -a                                  # عرض جميع الفروع المحلية والبعيدة على GitHub
git branch <branch_name>                       # إنشاء فرع جديد
git checkout <branch_name>                     # الانتقال إلى فرع محدد (طريقة قديمة)
git switch <branch_name>                       # الانتقال إلى فرع محدد (طريقة حديثة وموصى بها)
git checkout -b <branch_name>                  # إنشاء فرع جديد والانتقال إليه فوراً
git switch -c <branch_name>                    # إنشاء فرع جديد والانتقال إليه فوراً (طريقة حديثة)
git merge <branch_name>                        # دمج الفرع المترجم مع الفرع الحالي الذي تقف عليه
git branch -d <branch_name>                    # حذف فرع محلي بعد دمجه
git branch -D <branch_name>                    # إجبار حذف فرع محلي حتى لو لم يتم دمجه

# ==========================================
# 6. سجل التغييرات والتاريخ (History & Logs)
# ==========================================
git log                                        # عرض سجل عمليات الحفظ (Commits) بالتفصيل
git log --oneline                              # عرض سجل التغييرات بشكل مختصر (سطر واحد لكل Commit)
git diff                                       # مقارنة الملفات والتعديلات الحالية مع آخر Commit
git diff --staged                              # عرض التعديلات الموجودة في مرحلة Staging قبل عمل Commit

# ==========================================
# 7. التراجع والإلغاء (Undoing & Stashing)
# ==========================================
git restore <file_name>                        # التراجع عن التعديلات غير المحفوظة في ملف محدد
git restore --staged <file_name>               # سحب الملف من مرحلة Staging دون إلغاء التعديلات
git reset --soft HEAD~1                        # إلغاء آخر Commit مع الإبقاء على التعديلات في Staging
git reset --hard HEAD~1                        # إلغاء آخر Commit ومسح جميع التعديلات التابعة له نهائياً
git stash                                      # حفظ التعديلات المؤقتة جانباً وتفريغ مجلد العمل
git stash pop                                  # استعادة آخر تعديلات تم تخزينها مؤقتاً عبر stash
git stash list                                 # عرض قائمة بكل التخزينات المؤقتة


كيف تعرف أي إيميل مسجل حالياً؟
لمعرفة الإيميل المسجل عالمياً (على مستوى الجهاز بأكمله):

Bash
git config --global user.email

لمعرفة الإيميل المسجل للمشروع الحالي الذي تفتحه في التيرمنال:

Bash
git config user.email

لمعرفة اسم المستخدم المسجل:

Bash
git config user.name


1. تغيير بيانات الهوية (اسم صاحب الـ Commit وإيميله)
لتغيير الإيميل والاسم اللذين يُكتبان في الـ Commits (أي التحويل بين حساب أخر وحسابك الحالي):

تغيير الإيميل للمشروع الحالي:

Bash
git config user.email "your_new_email@example.com"
git config user.name "Your_New_Name"
تغيير الإيميل لجميع المشاريع:

Bash
git config --global user.email "your_new_email@example.com"
git config --global user.name "Your_New_Name"
2. تسجيل الخروج من حساب GitHub المخزن على الويندوز (Credentials)
عند رفع الكود لأول مرة (git push)، يحفظ ويندوز بيانات اعتمادك (Passwords / Tokens). لإلغائها أو تسجيل الخروج لحساب جديد:

افتح قائمة Start وابحث عن Credential Manager (مدير الاعتمادات).

اختر Windows Credentials.

ابحث ضمن القائمة عن git:[https://github.com](https://github.com).

اضغط عليها واختر Remove (حذف).

في المرة القادمة التي تقوم فيها بعمل git push أو git clone لمستودع خاص، سيطلب منك Git تسجيل الدخول مجدداً عبر نافذة متصفح تتطلب اسم المستخدم وكلمة السر الحساب الجديد.
